cmake_minimum_required(VERSION 3.30)
project(binding_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BENCH_MATCHED "Disable library-specific size/LTO/strip optimizations" ON)
set(BENCH_SCALE_COUNT 0 CACHE STRING "Generated bindings per module; 0 runs runtime suite")
set(BENCH_SCALE_KIND functions CACHE STRING "functions or classes")

if(NOT DEFINED ENV{CONDA_PREFIX} OR NOT Python_EXECUTABLE)
  message(FATAL_ERROR "Use pixi run configure/build/test/bench (Python must be explicit).")
endif()
find_package(Python 3.10 REQUIRED COMPONENTS Interpreter Development.Module)
file(REAL_PATH "$ENV{CONDA_PREFIX}" pixi_prefix)
foreach(p IN ITEMS "${Python_EXECUTABLE}" ${Python_INCLUDE_DIRS} ${Python_LIBRARIES})
  if(IS_ABSOLUTE "${p}")
    file(REAL_PATH "${p}" actual)
    cmake_path(IS_PREFIX pixi_prefix "${actual}" NORMALIZE in_pixi)
    if(NOT in_pixi)
      message(FATAL_ERROR "Python artifact outside pixi: ${actual}")
    endif()
  endif()
endforeach()
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(nanobind CONFIG REQUIRED)
message(STATUS "Bench Python: ${Python_EXECUTABLE} (${Python_VERSION})")
message(STATUS "Bench headers: ${Python_INCLUDE_DIRS}; libraries: ${Python_LIBRARIES}")

function(add_binding target backend source)
  if(backend STREQUAL "pybind11")
    if(BENCH_MATCHED)
      pybind11_add_module(${target} NO_EXTRAS ${source})
    else()
      pybind11_add_module(${target} ${source})
    endif()
  else()
    if(BENCH_MATCHED)
      nanobind_add_module(${target} NB_STATIC NOMINSIZE NOSTRIP ${source})
    else()
      nanobind_add_module(${target} NB_STATIC ${source})
    endif()
  endif()
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  if(MSVC)
    target_compile_options(${target} PRIVATE /bigobj)
  endif()
  set_target_properties(${target} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
endfunction()

if(BENCH_SCALE_COUNT GREATER 0)
  execute_process(COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_scale.py"
    --output "${CMAKE_BINARY_DIR}/generated" --count "${BENCH_SCALE_COUNT}" --kind "${BENCH_SCALE_KIND}"
    COMMAND_ERROR_IS_FATAL ANY)
  add_binding(scale_py pybind11 "${CMAKE_BINARY_DIR}/generated/py.cpp")
  add_binding(scale_nb nanobind "${CMAKE_BINARY_DIR}/generated/nb.cpp")
else()
  add_binding(bench_py pybind11 src/pybind.cpp)
  add_binding(bench_nb nanobind src/nanobind.cpp)
  execute_process(COMMAND "${Python_EXECUTABLE}" -c "import Cython; print(Cython.__version__)"
    OUTPUT_VARIABLE CYTHON_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  set(cython_source "${CMAKE_BINARY_DIR}/generated/bench_cy.cpp")
  add_custom_command(OUTPUT "${cython_source}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/generated"
    COMMAND "${Python_EXECUTABLE}" -m cython --cplus -3 --module-name bench_cy
      -o "${cython_source}" "${CMAKE_CURRENT_SOURCE_DIR}/src/cython_bindings.pyx"
    DEPENDS src/cython_bindings.pyx pixi.lock
    VERBATIM)
  Python_add_library(bench_cy MODULE WITH_SOABI "${cython_source}")
  target_include_directories(bench_cy PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  set_target_properties(bench_cy PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
  if(MSVC)
    target_compile_options(bench_cy PRIVATE /bigobj)
  endif()
  foreach(variant IN ITEMS py smart nb)
    if(variant STREQUAL "nb")
      set(backend nanobind)
    else()
      set(backend pybind11)
    endif()
    add_binding(sdk_${variant} ${backend} src/sdk.cpp)
    target_compile_definitions(sdk_${variant} PRIVATE SDK_NAMESPACE=sdk_${variant})
  endforeach()
  target_compile_definitions(sdk_nb PRIVATE SDK_NANOBIND)
  target_compile_definitions(sdk_smart PRIVATE SDK_SMART_HOLDER)
endif()
configure_file(cmake/build-info.json.in build-info.json @ONLY)
